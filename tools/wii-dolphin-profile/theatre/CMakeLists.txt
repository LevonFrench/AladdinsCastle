cmake_minimum_required(VERSION 3.20)
project(AladdinsCastleDolphinTheatre LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(PRODUCT_JSON "${CMAKE_CURRENT_SOURCE_DIR}/../../../data/product.json" ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PRODUCT_JSON}" "${CMAKE_CURRENT_SOURCE_DIR}/../product.py" "${CMAKE_CURRENT_SOURCE_DIR}/product_fixture.json")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../product.py"
  --product "${PRODUCT_JSON}" --header "${CMAKE_CURRENT_BINARY_DIR}/product_title.h"
  RESULT_VARIABLE PRODUCT_RESULT ERROR_VARIABLE PRODUCT_ERROR)
if(NOT PRODUCT_RESULT EQUAL 0)
  message(FATAL_ERROR "Invalid product labels: ${PRODUCT_ERROR}")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../product.py"
  --product "${CMAKE_CURRENT_SOURCE_DIR}/product_fixture.json" --header "${CMAKE_CURRENT_BINARY_DIR}/product_fixture.h"
  RESULT_VARIABLE PRODUCT_FIXTURE_RESULT)
if(NOT PRODUCT_FIXTURE_RESULT EQUAL 0)
  message(FATAL_ERROR "Product fixture generation failed")
endif()
add_executable(theatre-product-label-test product_label_test.cpp)
target_compile_features(theatre-product-label-test PRIVATE cxx_std_17)
target_include_directories(theatre-product-label-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME theatre-product-label-bytes COMMAND theatre-product-label-test)
add_executable(theatre-core-test core_test.cpp)
target_compile_features(theatre-core-test PRIVATE cxx_std_17)
add_test(NAME theatre-cpu COMMAND theatre-core-test)
add_executable(theatre-failure-diagnostic-test failure_diagnostic_test.cpp)
target_compile_features(theatre-failure-diagnostic-test PRIVATE cxx_std_17)
add_test(NAME theatre-failure-diagnostic-cpu COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_failure_json.py" "$<TARGET_FILE:theatre-failure-diagnostic-test>")
if(WIN32)
  set(OPENVR_SOURCE "" CACHE PATH "Explicit existing OpenVR SDK; no downloads")
  if(NOT EXISTS "${OPENVR_SOURCE}/headers/openvr.h")
    message(FATAL_ERROR "Provide the existing SDK with -DOPENVR_SOURCE")
  endif()
  add_executable(acvr-dolphin-theatre viewer.cpp)
  target_include_directories(acvr-dolphin-theatre PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  add_library(theatre-wgc STATIC wgc_capture.cpp)
  target_compile_features(theatre-wgc PUBLIC cxx_std_17)
  target_link_libraries(theatre-wgc PUBLIC theatre-d3d-upload windowsapp windowscodecs ole32 dwmapi)
  add_executable(theatre-wgc-test wgc_capture_test.cpp)
  target_link_libraries(theatre-wgc-test PRIVATE theatre-wgc)
  add_test(NAME theatre-software-wgc-readback COMMAND theatre-wgc-test "${CMAKE_CURRENT_BINARY_DIR}/synthetic-test-output")
  add_library(theatre-d3d-upload STATIC d3d_upload.cpp)
  target_compile_features(theatre-d3d-upload PUBLIC cxx_std_17)
  target_compile_definitions(theatre-d3d-upload PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(theatre-d3d-upload PUBLIC d3d11 dxgi)
  add_executable(theatre-d3d-upload-test d3d_upload_test.cpp)
  target_link_libraries(theatre-d3d-upload-test PRIVATE theatre-d3d-upload)
  add_test(NAME theatre-software-d3d-upload COMMAND theatre-d3d-upload-test)
  add_test(NAME theatre-reject-zero-pid COMMAND acvr-dolphin-theatre --pid 0)
  set_tests_properties(theatre-reject-zero-pid PROPERTIES WILL_FAIL TRUE)
  add_test(NAME theatre-reject-unbounded-run COMMAND acvr-dolphin-theatre --pid 12 --seconds 601)
  set_tests_properties(theatre-reject-unbounded-run PROPERTIES WILL_FAIL TRUE)
  add_test(NAME theatre-reject-unrecognized-option COMMAND acvr-dolphin-theatre --unrecognized 0)
  set_tests_properties(theatre-reject-unrecognized-option PROPERTIES WILL_FAIL TRUE)
  add_test(NAME theatre-reject-relative-failure-directory COMMAND acvr-dolphin-theatre --failure-diagnostic-dir relative)
  set_tests_properties(theatre-reject-relative-failure-directory PROPERTIES WILL_FAIL TRUE)
  add_test(NAME theatre-recenter-options-cpu COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_recenter_options.py" "$<TARGET_FILE:acvr-dolphin-theatre>")
  set_tests_properties(theatre-recenter-options-cpu PROPERTIES TIMEOUT 35)
  add_executable(theatre-dll-probe-test dll_probe_test.cpp)
  add_executable(theatre-dpi-test dpi_test.cpp)
  target_compile_features(theatre-dpi-test PRIVATE cxx_std_17)
  add_test(NAME theatre-physical-pixels COMMAND theatre-dpi-test)
  target_compile_features(theatre-dll-probe-test PRIVATE cxx_std_17)
  add_test(NAME theatre-dll-probe COMMAND theatre-dll-probe-test "${OPENVR_SOURCE}/bin/win64/openvr_api.dll")
  target_compile_features(acvr-dolphin-theatre PRIVATE cxx_std_17)
  target_compile_definitions(acvr-dolphin-theatre PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(acvr-dolphin-theatre PRIVATE "${OPENVR_SOURCE}/headers")
  target_link_libraries(acvr-dolphin-theatre PRIVATE theatre-wgc "${OPENVR_SOURCE}/lib/win64/openvr_api.lib" user32 dwmapi delayimp)
  target_link_options(acvr-dolphin-theatre PRIVATE /DELAYLOAD:openvr_api.dll)
  configure_file("${OPENVR_SOURCE}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/OpenVR-LICENSE.txt" COPYONLY)
  add_custom_command(TARGET acvr-dolphin-theatre POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${OPENVR_SOURCE}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:acvr-dolphin-theatre>")
