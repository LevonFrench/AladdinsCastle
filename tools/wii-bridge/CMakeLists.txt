cmake_minimum_required(VERSION 3.20)
project(AladdinsCastleWiiBridge LANGUAGES CXX)
set(OPENVR_SOURCE "" CACHE PATH "Existing pinned OpenVR checkout; never downloaded automatically")
if(NOT EXISTS "${OPENVR_SOURCE}/headers/openvr.h")
  message(FATAL_ERROR "Pass -DOPENVR_SOURCE pointing at the existing OpenVR SDK")
endif()
add_executable(acvr-wii-pose pose_feed.cpp)
target_compile_features(acvr-wii-pose PRIVATE cxx_std_17)
target_include_directories(acvr-wii-pose PRIVATE "${OPENVR_SOURCE}/headers")
if(WIN32)
  target_link_libraries(acvr-wii-pose PRIVATE "${OPENVR_SOURCE}/lib/win64/openvr_api.lib")
  add_custom_command(TARGET acvr-wii-pose POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${OPENVR_SOURCE}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:acvr-wii-pose>")
else()
  target_link_libraries(acvr-wii-pose PRIVATE "${OPENVR_SOURCE}/lib/linux64/libopenvr_api.so")
endif()
