cmake_minimum_required(VERSION 3.20)
project(acvr_gun_room LANGUAGES C CXX)
include(CTest)
if(NOT WIN32)
    message(FATAL_ERROR "Native gun room currently requires Windows desktop GL")
endif()
set(ACVR_BUILD_OPENXR_GL ON CACHE BOOL "" FORCE)
set(ACVR_BUILD_GL_GUN_RENDERER ON CACHE BOOL "" FORCE)
if(NOT ACVR_JSON_INCLUDE_DIR OR NOT ACVR_TOML_INCLUDE_DIR)
    message(FATAL_ERROR "Use already-local ACVR_JSON_INCLUDE_DIR and ACVR_TOML_INCLUDE_DIR")
endif()
add_subdirectory(../../libacvr acvr-build)
add_executable(acvr_gun_room main.cpp room_scene.cpp capture_plan.cpp control_stdin.cpp)
target_compile_features(acvr_gun_room PRIVATE cxx_std_17)
target_include_directories(acvr_gun_room PRIVATE ../../libacvr/include "${ACVR_JSON_INCLUDE_DIR}")
target_link_libraries(acvr_gun_room PRIVATE acvr opengl32)
if(MSVC)
    target_compile_options(acvr_gun_room PRIVATE /W4 /WX /permissive-)
endif()
add_executable(acvr_gun_room_cpu_tests room_scene.cpp room_scene_tests.cpp)
target_compile_features(acvr_gun_room_cpu_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(acvr_gun_room_cpu_tests PRIVATE /W4 /WX /permissive-)
endif()
add_test(NAME acvr_gun_room_math COMMAND acvr_gun_room_cpu_tests)
add_executable(acvr_gun_room_capture_plan_tests capture_plan.cpp capture_plan_tests.cpp)
target_compile_features(acvr_gun_room_capture_plan_tests PRIVATE cxx_std_17)
target_include_directories(acvr_gun_room_capture_plan_tests PRIVATE "${ACVR_JSON_INCLUDE_DIR}" ../../libacvr/src)
if(MSVC)
    target_compile_options(acvr_gun_room_capture_plan_tests PRIVATE /W4 /WX /permissive-)
endif()
add_test(NAME acvr_gun_room_capture_plan
         COMMAND acvr_gun_room_capture_plan_tests "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(acvr_gun_room_control_stdin_tests control_stdin.cpp control_stdin_tests.cpp)
target_compile_features(acvr_gun_room_control_stdin_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(acvr_gun_room_control_stdin_tests PRIVATE /W4 /WX /permissive-)
endif()
add_test(NAME acvr_gun_room_control_stdin COMMAND acvr_gun_room_control_stdin_tests)
set_tests_properties(acvr_gun_room_control_stdin PROPERTIES TIMEOUT 5)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME acvr_gun_room_cli
             COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/cli_tests.py"
                     "$<TARGET_FILE:acvr_gun_room>" "${CMAKE_CURRENT_BINARY_DIR}")
endif()
