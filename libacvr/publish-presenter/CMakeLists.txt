cmake_minimum_required(VERSION 3.20)
project(acvr_publish_presenter LANGUAGES CXX)
if(NOT WIN32 OR NOT EXISTS "${ACVR_OPENXR_INCLUDE_DIR}/openxr/openxr.h")
  message(FATAL_ERROR "Publish presenter requires Windows and existing ACVR_OPENXR_INCLUDE_DIR; no dependencies are fetched")
endif()
add_library(acvr_publish_presenter STATIC Presenter.cpp)
find_package(Threads REQUIRED)
target_compile_features(acvr_publish_presenter PUBLIC cxx_std_17)
target_include_directories(acvr_publish_presenter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${ACVR_OPENXR_INCLUDE_DIR}")
target_link_libraries(acvr_publish_presenter PUBLIC Threads::Threads opengl32 gdi32 user32)
