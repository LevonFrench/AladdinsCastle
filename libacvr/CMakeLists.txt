cmake_minimum_required(VERSION 3.20)
project(acvr_contract VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
find_package(Threads REQUIRED)
add_library(acvr STATIC src/runtime.cpp)
target_link_libraries(acvr PRIVATE Threads::Threads)
target_compile_features(acvr PUBLIC cxx_std_17)
target_include_directories(acvr PUBLIC include PRIVATE src)
if(MSVC)
    target_compile_options(acvr PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(acvr PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
# Optional native asset reader uses an existing header location, never a fetch.
set(ACVR_JSON_INCLUDE_DIR "" CACHE PATH "Existing nlohmann JSON include directory")
if(ACVR_JSON_INCLUDE_DIR)
    if(NOT EXISTS "${ACVR_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
        message(FATAL_ERROR "ACVR_JSON_INCLUDE_DIR does not contain nlohmann/json.hpp")
    endif()
    add_library(acvr_gun_assets STATIC src/gun_asset.cpp)
    target_compile_features(acvr_gun_assets PUBLIC cxx_std_17)
    target_include_directories(acvr_gun_assets PUBLIC src PRIVATE "${ACVR_JSON_INCLUDE_DIR}")
    if(MSVC)
        target_compile_options(acvr_gun_assets PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(acvr_gun_assets PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endif()
# No graphics, XR or third-party fetches. Real XR creation remains unsupported.
if(BUILD_TESTING)
    add_executable(acvr_contract_c tests/abi_contract.c)
    add_executable(acvr_contract_cpp tests/abi_contract.cpp)
    foreach(target acvr_contract_c acvr_contract_cpp)
        target_include_directories(${target} PRIVATE include)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
        add_test(NAME ${target} COMMAND ${target})
    endforeach()
    set_target_properties(acvr_contract_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    set_target_properties(acvr_contract_cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    add_executable(acvr_runtime_tests tests/runtime_tests.cpp)
    target_link_libraries(acvr_runtime_tests PRIVATE acvr)
    target_include_directories(acvr_runtime_tests PRIVATE src)
    if(MSVC)
        target_compile_options(acvr_runtime_tests PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(acvr_runtime_tests PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME acvr_runtime_core COMMAND acvr_runtime_tests)
    if(TARGET acvr_gun_assets)
        add_executable(acvr_gun_asset_tests tests/gun_asset_tests.cpp)
        target_link_libraries(acvr_gun_asset_tests PRIVATE acvr_gun_assets)
        target_include_directories(acvr_gun_asset_tests PRIVATE "${ACVR_JSON_INCLUDE_DIR}")
        if(MSVC)
            target_compile_options(acvr_gun_asset_tests PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(acvr_gun_asset_tests PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
        add_test(NAME acvr_gun_assets COMMAND acvr_gun_asset_tests)
    endif()
endif()
