cmake_minimum_required(VERSION 3.20)
project(acvr_contract VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
add_library(acvr STATIC src/runtime.cpp src/view_math.cpp)
target_link_libraries(acvr PRIVATE Threads::Threads)
target_compile_features(acvr PUBLIC cxx_std_17)
target_include_directories(acvr PUBLIC include PRIVATE src)
if(MSVC)
    target_compile_options(acvr PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(acvr PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
# Optional native asset reader uses an existing header location, never a fetch.
set(ACVR_JSON_INCLUDE_DIR "" CACHE PATH "Existing nlohmann JSON include directory")
if(ACVR_JSON_INCLUDE_DIR)
    if(NOT EXISTS "${ACVR_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
        message(FATAL_ERROR "ACVR_JSON_INCLUDE_DIR does not contain nlohmann/json.hpp")
    endif()
    add_library(acvr_gun_assets STATIC src/gun_asset.cpp)
    target_compile_features(acvr_gun_assets PUBLIC cxx_std_17)
    target_include_directories(acvr_gun_assets PUBLIC src PRIVATE "${ACVR_JSON_INCLUDE_DIR}")
    if(MSVC)
        target_compile_options(acvr_gun_assets PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(acvr_gun_assets PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endif()
# No graphics, XR or third-party fetches. Real XR creation remains unsupported.
set(ACVR_TOML_INCLUDE_DIR "" CACHE PATH "Existing tomlplusplus include directory")
if(ACVR_TOML_INCLUDE_DIR)
    if(NOT TARGET acvr_gun_assets OR NOT EXISTS "${ACVR_TOML_INCLUDE_DIR}/toml++/toml.hpp")
        message(FATAL_ERROR "Gun models require valid existing JSON and TOML include directories")
    endif()
    add_library(acvr_gun_models STATIC src/gun_model.cpp src/controls.cpp)
    target_compile_features(acvr_gun_models PUBLIC cxx_std_17)
    target_include_directories(acvr_gun_models PUBLIC src include PRIVATE "${ACVR_TOML_INCLUDE_DIR}")
    target_link_libraries(acvr_gun_models PUBLIC acvr_gun_assets)
    if(MSVC)
        target_compile_options(acvr_gun_models PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(acvr_gun_models PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    target_link_libraries(acvr PRIVATE acvr_gun_models)
    target_compile_definitions(acvr PUBLIC ACVR_GUN_MODELS=1)
endif()
if(BUILD_TESTING)
    add_executable(acvr_contract_c tests/abi_contract.c)
    add_executable(acvr_contract_cpp tests/abi_contract.cpp)
    foreach(target acvr_contract_c acvr_contract_cpp)
        target_include_directories(${target} PRIVATE include)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
        add_test(NAME ${target} COMMAND ${target})
    endforeach()
    set_target_properties(acvr_contract_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    set_target_properties(acvr_contract_cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    add_executable(acvr_runtime_tests tests/runtime_tests.cpp)
    target_link_libraries(acvr_runtime_tests PRIVATE acvr)
    target_include_directories(acvr_runtime_tests PRIVATE src)
    if(TARGET acvr_gun_models)
        target_include_directories(acvr_runtime_tests PRIVATE "${ACVR_JSON_INCLUDE_DIR}")
    endif()
    if(MSVC)
        target_compile_options(acvr_runtime_tests PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(acvr_runtime_tests PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME acvr_runtime_core COMMAND acvr_runtime_tests)
    add_executable(acvr_view_math_tests tests/view_math_tests.cpp)
    target_link_libraries(acvr_view_math_tests PRIVATE acvr)
    target_include_directories(acvr_view_math_tests PRIVATE src)
    if(MSVC)
        target_compile_options(acvr_view_math_tests PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(acvr_view_math_tests PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME acvr_provider_view_math COMMAND acvr_view_math_tests)
    if(TARGET acvr_gun_assets)
        add_executable(acvr_gun_asset_tests tests/gun_asset_tests.cpp)
        target_link_libraries(acvr_gun_asset_tests PRIVATE acvr_gun_assets)
        target_include_directories(acvr_gun_asset_tests PRIVATE "${ACVR_JSON_INCLUDE_DIR}")
        if(MSVC)
            target_compile_options(acvr_gun_asset_tests PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(acvr_gun_asset_tests PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
        add_test(NAME acvr_gun_assets COMMAND acvr_gun_asset_tests)
    endif()
    if(TARGET acvr_gun_models)
        add_executable(acvr_gun_model_tests tests/gun_model_tests.cpp)
        target_link_libraries(acvr_gun_model_tests PRIVATE acvr_gun_models)
        target_include_directories(acvr_gun_model_tests PRIVATE "${ACVR_JSON_INCLUDE_DIR}")
        if(MSVC)
            target_compile_options(acvr_gun_model_tests PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(acvr_gun_model_tests PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
        add_test(NAME acvr_gun_models COMMAND acvr_gun_model_tests)
        add_executable(acvr_controls_tests tests/controls_tests.cpp)
        target_link_libraries(acvr_controls_tests PRIVATE acvr_gun_models)
        if(MSVC)
            target_compile_options(acvr_controls_tests PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(acvr_controls_tests PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
        add_test(NAME acvr_controls COMMAND acvr_controls_tests)
    endif()
endif()
option(ACVR_BUILD_GL_GUN_RENDERER "Build desktop GL gun renderer with installed GL headers" OFF)
if(ACVR_BUILD_GL_GUN_RENDERER)
    if(NOT TARGET acvr_gun_models)
        message(FATAL_ERROR "GL gun renderer requires existing JSON/TOML includes")
    endif()
    add_library(acvr_gun_gl STATIC src/gun_gl.cpp)
    target_link_libraries(acvr_gun_gl PUBLIC acvr_gun_models)
    if(MSVC)
        target_compile_options(acvr_gun_gl PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(acvr_gun_gl PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    if(BUILD_TESTING)
        add_executable(acvr_gun_gl_tests tests/gun_gl_tests.cpp)
        target_link_libraries(acvr_gun_gl_tests PRIVATE acvr_gun_gl)
        target_include_directories(acvr_gun_gl_tests PRIVATE "${ACVR_JSON_INCLUDE_DIR}")
        if(MSVC)
            target_compile_options(acvr_gun_gl_tests PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(acvr_gun_gl_tests PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
        add_test(NAME acvr_gun_gl_mock COMMAND acvr_gun_gl_tests)
    endif()
    option(ACVR_BUILD_GPU_SMOKE "Build manual owner-approved Windows synthetic GPU check (never CTest)" OFF)
    if(ACVR_BUILD_GPU_SMOKE)
        if(NOT WIN32)
            message(FATAL_ERROR "This manual context bootstrap is Windows-only")
        endif()
        find_package(Qt6 6.8 REQUIRED COMPONENTS Gui OpenGL)
        add_executable(acvr_gun_gl_smoke tests/gun_gl_smoke.cpp)
        target_link_libraries(acvr_gun_gl_smoke PRIVATE acvr_gun_gl Qt6::Gui Qt6::OpenGL opengl32)
        if(MSVC)
            target_compile_options(acvr_gun_gl_smoke PRIVATE /W4 /WX /permissive-)
        endif()
    endif()
endif()
