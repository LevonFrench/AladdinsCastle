cmake_minimum_required(VERSION 3.20)
project(Namco22NativeSlice LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(N22_OVERLAY "" CACHE PATH "Fresh ignored overlay generated by prepare.py")
set(SDL2_ROOT "" CACHE PATH "Locally unpacked approved SDL2 SDK")
set(N22_ACVR_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../libacvr/include" CACHE PATH "Reviewed libacvr headers; prepared contract may be explicitly supplied")
if(NOT N22_OVERLAY OR NOT SDL2_ROOT)
  message(FATAL_ERROR "Explicit N22_OVERLAY and SDL2_ROOT required")
endif()

set(units geo_hw ss22_gl eng slave_list tex_bake render_target quad_gl hud_edges post_gl fog_hw sprite_hw text_hw)
foreach(unit IN LISTS units)
  list(APPEND sources "${N22_OVERLAY}/engine/${unit}.c")
endforeach()
add_library(n22_native STATIC ${sources} eye.c pose_stage.c native_batch.c)
set_source_files_properties(native_batch.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
target_compile_definitions(n22_native PRIVATE ACN22_NATIVE_GPU_PROOF=0)
set_source_files_properties(pose_stage.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
set_source_files_properties(eye.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
target_include_directories(n22_native PUBLIC . ${N22_OVERLAY}/engine ${N22_ACVR_INCLUDE} ${SDL2_ROOT}/include)
target_compile_definitions(n22_native PRIVATE SDL_MAIN_HANDLED ACN22_SYNTHETIC_SLICE)
target_compile_options(n22_native PRIVATE -fno-strict-aliasing -fwrapv -ffunction-sections -fdata-sections)
if(WIN32)
  target_compile_options(n22_native PRIVATE -include ${N22_OVERLAY}/include/win_compat.h)
endif()
target_link_directories(n22_native PUBLIC ${SDL2_ROOT}/lib)
target_link_libraries(n22_native PUBLIC SDL2 opengl32 m)
add_executable(n22_native_fixture fixture.c)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../../data/product.json" ac_product_json)
string(JSON ac_display_name GET "${ac_product_json}" display_name)
string(REPLACE "\\" "\\\\" ac_display_name "${ac_display_name}")
string(REPLACE "\"" "\\\"" ac_display_name "${ac_display_name}")
configure_file(product_title.h.in "${CMAKE_CURRENT_BINARY_DIR}/product_title.h" @ONLY)
target_include_directories(n22_native_fixture PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(n22_native_fixture PRIVATE n22_native)
target_compile_definitions(n22_native_fixture PRIVATE SDL_MAIN_HANDLED)
target_compile_options(n22_native_fixture PRIVATE -Wall -Wextra -Werror)
target_compile_options(n22_native PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wno-unused-function>)
target_link_options(n22_native_fixture PRIVATE -Wl,--gc-sections)
enable_testing()
add_test(NAME native_fixture_cpu COMMAND n22_native_fixture --check)
set_tests_properties(native_fixture_cpu PROPERTIES TIMEOUT 15)
add_executable(n22_pose_tests pose_tests.c)
target_link_libraries(n22_pose_tests PRIVATE n22_native)
target_compile_options(n22_pose_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME headpose_clipping_cpu COMMAND n22_pose_tests)
set_tests_properties(headpose_clipping_cpu PROPERTIES TIMEOUT 15)
add_executable(n22_native_batch_tests native_batch_tests.c)
target_link_libraries(n22_native_batch_tests PRIVATE n22_native)
target_compile_options(n22_native_batch_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME native_batch_cpu COMMAND n22_native_batch_tests)
set_tests_properties(native_batch_cpu PROPERTIES TIMEOUT 15)

# Separate source proof, omitted by default. Enable only after independent review
# and a separately coordinated native-build slot; enabling is not run authority.
option(ACN22_BUILD_PAIRED_PROOF "Build reviewed isolated synthetic proof executable" OFF)
if(ACN22_BUILD_PAIRED_PROOF)
  if(NOT WIN32)
    message(FATAL_ERROR "Paired proof protocol currently requires Windows owned pipes")
  endif()
  add_library(n22_paired_proof STATIC ${sources} eye.c pose_stage.c native_batch.c)
  target_include_directories(n22_paired_proof PUBLIC . ${N22_OVERLAY}/engine ${N22_ACVR_INCLUDE} ${SDL2_ROOT}/include)
  target_compile_definitions(n22_paired_proof PRIVATE ACN22_NATIVE_GPU_PROOF=1 SDL_MAIN_HANDLED ACN22_SYNTHETIC_SLICE)
  target_compile_options(n22_paired_proof PRIVATE -fno-strict-aliasing -fwrapv -ffunction-sections -fdata-sections
    -include ${N22_OVERLAY}/include/win_compat.h -Wno-unused-function)
  target_link_directories(n22_paired_proof PUBLIC ${SDL2_ROOT}/lib)
  target_link_libraries(n22_paired_proof PUBLIC SDL2 opengl32 m)
  add_executable(n22_native_batch_proof paired_native.c)
  target_include_directories(n22_native_batch_proof PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(n22_native_batch_proof PRIVATE ACN22_NATIVE_GPU_PROOF=1 SDL_MAIN_HANDLED)
  target_compile_options(n22_native_batch_proof PRIVATE -Wall -Wextra -Werror)
  target_link_options(n22_native_batch_proof PRIVATE -Wl,--gc-sections)
  target_link_libraries(n22_native_batch_proof PRIVATE n22_paired_proof)
endif()

# Adapter development remains separate from the accepted SDL proof and public query.
option(ACN22_BUILD_NATIVE_ADAPTER "Build isolated reviewed-header native adapter" OFF)
if(ACN22_BUILD_NATIVE_ADAPTER)
  if(NOT WIN32)
    message(FATAL_ERROR "Borrowed-context native adapter currently requires Windows")
  endif()
  file(READ "${N22_ACVR_INCLUDE}/acvr.h" acn22_header)
  if(NOT acn22_header MATCHES "ACVR_BACKEND_API_V1_SIZE")
    message(FATAL_ERROR "Explicit reviewed prepare-display header required")
  endif()
  add_library(n22_native_adapter STATIC ${sources} eye.c pose_stage.c native_batch.c
    native_display.c native_fixture.c native_adapter.c native_target_state.c)
  target_include_directories(n22_native_adapter PUBLIC . ../include ${N22_OVERLAY}/engine ${N22_ACVR_INCLUDE} ${SDL2_ROOT}/include)
  target_compile_definitions(n22_native_adapter PRIVATE ACN22_NATIVE_ADAPTER=1 ACN22_NATIVE_GPU_PROOF=1 SDL_MAIN_HANDLED ACN22_SYNTHETIC_SLICE)
  target_compile_options(n22_native_adapter PRIVATE -fno-strict-aliasing -fwrapv -ffunction-sections -fdata-sections
    -include ${N22_OVERLAY}/include/win_compat.h -Wno-unused-function)
  set_source_files_properties(native_adapter.c native_display.c native_target_state.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
  set_source_files_properties(native_fixture.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror;-Wno-unused-const-variable")
  target_link_directories(n22_native_adapter PUBLIC ${SDL2_ROOT}/lib)
  target_link_libraries(n22_native_adapter PUBLIC SDL2 opengl32 m)
  add_executable(n22_native_display_tests native_display_tests.c native_display.c native_fixture.c)
  target_link_libraries(n22_native_display_tests PRIVATE n22_native)
  target_compile_options(n22_native_display_tests PRIVATE -Wall -Wextra -Werror -Wno-unused-const-variable)
  add_test(NAME native_display_cpu COMMAND n22_native_display_tests)
  add_executable(n22_native_adapter_factory_tests native_adapter_factory_tests.c)
  target_link_libraries(n22_native_adapter_factory_tests PRIVATE n22_native_adapter)
  target_compile_options(n22_native_adapter_factory_tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME native_adapter_factory_cpu COMMAND n22_native_adapter_factory_tests)
  add_executable(n22_native_target_state_tests native_target_state_tests.c native_target_state.c)
  target_include_directories(n22_native_target_state_tests PRIVATE ${N22_ACVR_INCLUDE})
  target_compile_options(n22_native_target_state_tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME native_target_state_cpu COMMAND n22_native_target_state_tests)
  set_tests_properties(native_display_cpu native_adapter_factory_cpu native_target_state_cpu PROPERTIES TIMEOUT 15)
endif()
