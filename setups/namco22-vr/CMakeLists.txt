function(n22_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()

set(N22_ENGINE_SOURCE "" CACHE PATH "Explicit read-only pinned engine source for header-conformance tests")
if(N22_ENGINE_SOURCE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/patch_engine.py"
    --source "${N22_ENGINE_SOURCE}" RESULT_VARIABLE n22_guard_result)
  if(NOT n22_guard_result EQUAL 0)
    message(FATAL_ERROR "Pinned engine source guard failed")
  endif()
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_material_source.py"
    --source "${N22_ENGINE_SOURCE}" RESULT_VARIABLE n22_material_guard_result)
  if(NOT n22_material_guard_result EQUAL 0)
    message(FATAL_ERROR "Pinned material evidence guard failed")
  endif()
  add_library(namco22_source_bridge STATIC src/upstream_bridge.cpp)
  target_include_directories(namco22_source_bridge PUBLIC "${N22_ENGINE_SOURCE}/engine")
  target_link_libraries(namco22_source_bridge PUBLIC namco22_worker namco22_fog namco22_background namco22_polygon_fade namco22_composition)
