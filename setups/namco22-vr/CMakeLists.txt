cmake_minimum_required(VERSION 3.20)
project(Namco22StereoCPU LANGUAGES CXX)
include(CTest)
add_library(namco22_stereo_cpu STATIC src/scene.cpp src/backend.cpp)
target_compile_features(namco22_stereo_cpu PUBLIC cxx_std_17)
target_include_directories(namco22_stereo_cpu PUBLIC include ../../libacvr/include)
function(n22_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
n22_warnings(namco22_stereo_cpu)
add_executable(namco22_synthetic_capture tools/synthetic_capture.cpp)
target_link_libraries(namco22_synthetic_capture PRIVATE namco22_stereo_cpu)
n22_warnings(namco22_synthetic_capture)
if(BUILD_TESTING)
  add_executable(namco22_stereo_tests tests/stereo_tests.cpp)
  target_link_libraries(namco22_stereo_tests PRIVATE namco22_stereo_cpu)
  n22_warnings(namco22_stereo_tests)
  add_test(NAME namco22_stereo_synthetic COMMAND namco22_stereo_tests)
endif()
